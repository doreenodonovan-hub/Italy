function hav(a,b){const r=x=>x*Math.PI/180,R=6371,dlat=r(b[0]-a[0]),dlon=r(b[1]-a[1]),q=Math.sin(dlat/2)**2+Math.cos(r(a[0]))*Math.cos(r(b[0]))*Math.sin(dlon/2)**2;return 2*R*Math.asin(Math.sqrt(q))}
